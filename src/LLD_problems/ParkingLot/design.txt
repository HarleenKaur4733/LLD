interface FeeStrategy{

	double calculateCost();

}


class HourBasedFeeStrategy implements FeeStrategy{
	
	int hours;
	Ticket ticket;
	
	@Override
	double calculateCost(){
		return (ticket.getExitTime() - ticket.getEnteryTime()) * hours;
	}

}


class SizeBasedFeeStrategy implements FeeStrategy{
	
	int smallVehicleFee;
	int mediumVehicleFee;
	int largeVehicleFree;
	
	Ticket ticket;
	
	@Override
	double calculateCost(){
		VehicleSize vehicleSize = ticket.getVehicle().getSize();
		SWITCH(vehcileSize){
			case SMALL:
				returm smallVehicleFee;
			case MEDIUM:
				return mediumVehicleFee;
			case LARGE:
				return largeVehicleFree;
			default:
				throw new IllegalArgumentException("Unknown vehicle size: " + vehicleSize);
		
		}
	}

}
